cmake_minimum_required(VERSION 3.20)
project(ember_orb_rgbd LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(DEFINED OPENCV_PREFIX)
  if(NOT IS_ABSOLUTE "${OPENCV_PREFIX}")
    message(FATAL_ERROR "OPENCV_PREFIX must be an absolute installation")
  endif()
  set(OpenCV_DIR "${OPENCV_PREFIX}/lib/cmake/opencv5")
endif()
find_package(ORB_SLAM3 1.0.0 EXACT CONFIG REQUIRED)
find_package(PNG 1.6.58 EXACT REQUIRED)
add_executable(orb-rgbd main.cpp dataset.cpp png-reader.c)
target_compile_options(orb-rgbd PRIVATE -Wall -Wextra)
target_link_libraries(orb-rgbd PRIVATE ORB_SLAM3::ORB_SLAM3 PNG::PNG)
enable_testing()
add_test(NAME dataset-math COMMAND orb-rgbd --self-test)
set_tests_properties(dataset-math PROPERTIES TIMEOUT 10)
