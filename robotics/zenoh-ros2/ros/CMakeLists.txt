cmake_minimum_required(VERSION 3.16)
project(ember_zenoh_bridge LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(zenohpico 1.10.1 EXACT CONFIG REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/State.msg" "msg/Command.msg" "msg/Ack.msg")
rosidl_get_typesupport_target(message_support ${PROJECT_NAME} "rosidl_typesupport_cpp")
add_library(robot_transport STATIC ../protocol.c ../link.c)
target_include_directories(robot_transport PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_link_libraries(robot_transport PUBLIC zenohpico::lib)
foreach(program bridge probe)
  add_executable(robot-${program} ${program}.cpp)
  ament_target_dependencies(robot-${program} rclcpp)
  target_link_libraries(robot-${program} robot_transport "${message_support}")
  install(TARGETS robot-${program} DESTINATION lib/${PROJECT_NAME})
endforeach()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
