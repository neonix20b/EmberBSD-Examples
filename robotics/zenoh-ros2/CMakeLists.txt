cmake_minimum_required(VERSION 3.16)
project(ember_robot_controller LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(zenohpico 1.10.1 EXACT CONFIG REQUIRED)
add_library(robot-protocol protocol.c)
target_include_directories(robot-protocol PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(robot-link link.c)
target_link_libraries(robot-link PUBLIC robot-protocol zenohpico::lib)
add_executable(robot-controller controller.c)
target_link_libraries(robot-controller PRIVATE robot-link Threads::Threads)
enable_testing()
add_executable(test-protocol tests/test_protocol.c)
target_link_libraries(test-protocol PRIVATE robot-protocol)
target_compile_options(test-protocol PRIVATE -UNDEBUG)
add_test(NAME protocol COMMAND test-protocol)
add_executable(test-tcp-timeout tests/test_tcp_timeout.c)
target_link_libraries(test-tcp-timeout PRIVATE zenohpico::lib)
target_compile_options(test-tcp-timeout PRIVATE -UNDEBUG)
add_test(NAME tcp-connect-timeout COMMAND test-tcp-timeout connect)
add_test(NAME tcp-accept-timeout COMMAND test-tcp-timeout accept)
set_tests_properties(tcp-connect-timeout tcp-accept-timeout PROPERTIES TIMEOUT 8)
